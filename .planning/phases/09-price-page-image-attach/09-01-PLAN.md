      // PRICE-01: 평당가 전역 저장 — 검증 모달 완료 시 최신값으로 갱신
      (function(){
        const areaEl = document.getElementById('cArea');
        const totalArea = areaEl ? parseFloat(areaEl.value) || 0 : 0;
        if (totalArea > 0) {
          const finalCost = cart.filter(c => !c.gift).reduce((s, c) => s + c.sp * c.qty, 0);
