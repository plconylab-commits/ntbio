    ```javascript
    // ─── 이미지 첨부 IndexedDB 함수 (ATTACH-02) ──────────────────
    function _idbSaveImage(rec) {
      return new Promise((resolve, reject) => {
        const req = indexedDB.open(IDB_NAME, IDB_VER);
        req.onsuccess = (e) => {
          const db = e.target.result;
          const tx = db.transaction('attachImages', 'readwrite');
          tx.objectStore('attachImages').put(rec);
          tx.oncomplete = () => resolve();
          tx.onerror = () => reject(tx.error);
        };
        req.onerror = () => reject(req.error);
      });
    }

    function _idbLoadImages() {
      return new Promise((resolve, reject) => {
        const req = indexedDB.open(IDB_NAME, IDB_VER);
        req.onsuccess = (e) => {
          const db = e.target.result;
          const tx = db.transaction('attachImages', 'readonly');
          const all = tx.objectStore('attachImages').getAll();
          all.onsuccess = () => resolve(all.result || []);
          all.onerror = () => reject(all.error);
        };
        req.onerror = () => reject(req.error);
      });
    }

    function _idbDeleteImage(id) {
      return new Promise((resolve, reject) => {
        const req = indexedDB.open(IDB_NAME, IDB_VER);
        req.onsuccess = (e) => {
          const db = e.target.result;
          const tx = db.transaction('attachImages', 'readwrite');
          tx.objectStore('attachImages').delete(id);
          tx.oncomplete = () => resolve();
          tx.onerror = () => reject(tx.error);
        };
        req.onerror = () => reject(req.error);
      });
    }
    ```

    ```javascript
    function onImgSelected(input) {
      const files = Array.from(input.files);
      if (!files.length) return;
      let pending = 0;
      files.forEach(file => {
        if (file.size > 5 * 1024 * 1024) {
          alert(`${file.name}: 5MB 이하 이미지만 첨부 가능합니다.`);
          return;
        }
        pending++;
        const reader = new FileReader();
        reader.onload = (e) => {
          const rec = {
            id: 'img_' + Date.now() + '_' + Math.random().toString(36).slice(2),
            name: file.name,
            dataUrl: e.target.result,
            savedAt: new Date().toISOString(),
            memo: ''
          };
          _idbSaveImage(rec).then(() => {
            _attachedImages.push(rec);
            pending--;
            if (pending === 0) syncPrint();
          }).catch(err => {
            console.error('[이미지 저장 실패]', err);
            pending--;
          });
        };
        reader.readAsDataURL(file);
      });
      input.value = '';  // 동일 파일 재선택 허용
    }

    function openImgManager() {
      _idbLoadImages().then(imgs => {
        _attachedImages = imgs;
        const list = imgs.length === 0
          ? '<p style="color:#888;text-align:center;padding:20px;">첨부된 이미지가 없습니다.</p>'
          : imgs.map(img => `
            <div style="display:flex;align-items:center;gap:10px;padding:8px 0;border-bottom:1px solid #eee;">
              <img src="${img.dataUrl}" style="width:60px;height:60px;object-fit:cover;border-radius:4px;border:1px solid #ddd;">
              <div style="flex:1;font-size:13px;">
                <div style="font-weight:700;">${img.name}</div>
                <div style="color:#888;font-size:11px;">${img.savedAt ? img.savedAt.slice(0,10) : ''}</div>
              </div>
              <button onclick="deleteAttachedImage('${img.id}')"
                style="background:#FFEBEE;color:#C62828;border:1px solid #EF9A9A;border-radius:5px;padding:4px 10px;cursor:pointer;font-size:12px;">삭제</button>
            </div>`).join('');
        document.getElementById('imgManagerList').innerHTML = list;
        document.getElementById('imgManagerOverlay').style.display = 'flex';
        document.getElementById('overlay').style.display = 'block';
      });
    }

    function deleteAttachedImage(id) {
      _idbDeleteImage(id).then(() => {
        _attachedImages = _attachedImages.filter(i => i.id !== id);
        syncPrint();
        openImgManager();  // 목록 갱신
      }).catch(err => console.error('[이미지 삭제 실패]', err));
    }

    function closeImgManager() {
      document.getElementById('imgManagerOverlay').style.display = 'none';
      document.getElementById('overlay').style.display = 'none';
    }
    ```
