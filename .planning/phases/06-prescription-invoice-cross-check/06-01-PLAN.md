    ```javascript
    (function() {
      'use strict';

      function normalizeForMatch(str) {
        return (str || '').toLowerCase().replace(/[^가-힣a-z0-9]/g, '');
      }

      function fuzzyMatch(a, b) {
        const na = normalizeForMatch(a);
        const nb = normalizeForMatch(b);
        if (!na || !nb) return false;
        return na.includes(nb) || nb.includes(na);
      }

      function extractRxItems(prescriptionJSON) {
        const totalArea = prescriptionJSON.farmInfo?.totalArea || null;
        const rows = prescriptionJSON.rxRows || [];
        return rows
          .filter(r => r.productName && (r.dosageQty > 0 || r.baseQty > 0))
          .map(r => ({
            name: r.productName,
            rawName: r.productRaw || r.originalText || '',
            qty: _calcRequiredQty(totalArea, r.baseArea, r.dosageQty || r.baseQty),
            baseQty: r.dosageQty || r.baseQty,
            baseArea: r.baseArea,
          }));
      }

      // Inline calcRequiredQty to avoid dependency on prescriptionModel.js in test context
      function _calcRequiredQty(totalArea, baseArea, baseQty) {
        if (!baseArea || !totalArea) return baseQty;
        return Math.ceil((totalArea / baseArea) * baseQty);
      }

      function buildDiffRows(rxItems, cartItems) {
        // 1:1 greedy matching — sort by normalized name length DESC (longer = more specific first)
        const rxSorted = rxItems.slice().sort((a, b) =>
          normalizeForMatch(b.name).length - normalizeForMatch(a.name).length
        );
        const cartUsed = new Set();
        const rows = [];

        // Match rx items to cart items
        for (const rx of rxSorted) {
          let bestIdx = -1;
          let bestLen = 0;
          for (let i = 0; i < cartItems.length; i++) {
            if (cartUsed.has(i)) continue;
            if (fuzzyMatch(rx.name, cartItems[i].name)) {
              const len = normalizeForMatch(cartItems[i].name).length;
              if (len > bestLen) { bestLen = len; bestIdx = i; }
            }
          }
          if (bestIdx >= 0) {
            cartUsed.add(bestIdx);
            const ci = cartItems[bestIdx];
            const status = rx.qty === ci.qty ? 'match' : 'qty-diff';
            rows.push({
              status,
              rxName: rx.rawName || rx.name,
              rxQty: rx.qty,
              cartName: ci.name,
              cartQty: ci.qty,
              cartAmount: (ci.sp || 0) * (ci.qty || 0),
              diff: rx.qty - ci.qty,
            });
          } else {
            rows.push({
              status: 'rx-only',
              rxName: rx.rawName || rx.name,
              rxQty: rx.qty,
              cartName: null,
              cartQty: null,
              cartAmount: null,
              diff: null,
            });
          }
        }

        // Remaining cart items not matched
        for (let i = 0; i < cartItems.length; i++) {
          if (cartUsed.has(i)) continue;
          const ci = cartItems[i];
          rows.push({
            status: 'cart-only',
            rxName: null,
            rxQty: null,
            cartName: ci.name,
            cartQty: ci.qty,
            cartAmount: (ci.sp || 0) * (ci.qty || 0),
            diff: null,
          });
        }

        return rows;
      }

      function calcCartPricePerPyeong(cartItems, area) {
        if (!area || area <= 0) return null;
        const total = cartItems.reduce((sum, item) => sum + (item.sp || 0) * (item.qty || 0), 0);
        return Math.round(total / area);
      }

      // Expose for browser (window) and test (vm context)
      if (typeof window !== 'undefined') {
        window.RxCompare = { normalizeForMatch, fuzzyMatch, buildDiffRows, calcCartPricePerPyeong, extractRxItems };
      } else {
        // Node.js test context — expose to context globals
        normalizeForMatch; fuzzyMatch; buildDiffRows; calcCartPricePerPyeong; extractRxItems;
      }
    })();
    ```

    NOTE on exports for test context: The IIFE pattern used by rxNormalizer.js exposes individual functions at the context level. Check how rxNormalizer.v14.test.js extracts functions — rxCompare.js must expose functions the same way. The pattern is: declare functions with `function name(){}` at the IIFE top-level scope. When `vm.runInContext` runs the IIFE, the functions declared inside it are NOT accessible from ctx unless explicitly assigned. Use the same trick as rxNormalizer.js: assign each function to the context scope OUTSIDE the IIFE, OR use a non-IIFE approach for testability. Read rxNormalizer.js to see how it handles this.
