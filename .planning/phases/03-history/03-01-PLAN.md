```javascript
/** 처방이력 스냅샷 빌드 — doPrint/doPdf/doEmail에서 호출 */
function _buildPrescrSnapshot(){
  if(!cart.length) return null;
  var custId = (typeof CustomerUI !== 'undefined' && CustomerUI.getCurrentCustomerId)
    ? CustomerUI.getCurrentCustomerId() : null;
  return {
    customerId: custId,
    customer: {
      name:   (document.getElementById('cName')   || {}).value || '',
      crop:   (document.getElementById('cCrop')   || {}).value || '',
      area:   parseFloat((document.getElementById('cArea') || {}).value) || 0,
      region: (document.getElementById('cRegion') || {}).value || '',
      phone:  (document.getElementById('cPhone')  || {}).value || ''
    },
    items: cart.map(function(c){
      return { name:c.name, size:c.size, retail:c.retail, sp:c.sp, qty:c.qty, gift:c.gift||false };
    }),
    discountRate: parseFloat((document.getElementById('gDisc') || {}).value) || 0,
    vatIncluded: (document.getElementById('vatChk') || {}).checked || false,
    totals: {
      supply:     cart.filter(function(c){return !c.gift;}).reduce(function(s,c){return s+c.sp*c.qty;},0),
      grandTotal: Math.round(cart.filter(function(c){return !c.gift;}).reduce(function(s,c){return s+c.sp*c.qty;},0) * ((document.getElementById('vatChk')||{}).checked ? 1.1 : 1))
    }
  };
}
```
